type allocator interface {
	AllocDense(uint64) uint64
	AllocBatch(uint64,int32) uint64
	AddActiveCount(uint64,int32)
	PopBatch(int,int)(uint64,uint64,int)
	Alloc(uint64) uint64
	Free(uint64)
	GetPtr(uint64) unsafe.Pointer
	Close() error
}

var backends = []struct {
	name string
	new  func(uint64) (allocator, error)
}{
	{"original", func(size uint64) (allocator, error) { return utils.NewArena[uint64](size, -1) }},
	{"experimental", func(size uint64) (allocator, error) { return arena.New(size) }},
	{"shared",func(size uint64)(allocator,error){return shared.NewArena[uint64](size,-1)}},
	{"nonmoving",func(size uint64)(allocator,error){return nonmoving.New(size)}},
}

// Both sides use the same interface overhead. These sizes already have reusable
// classes in utils; a pinned live block keeps the region from being purged.
func BenchmarkHotReuse(b *testing.B) {
	for _, size := range []uint64{16, 64, 4096} {
		for _, backend := range backends {
			b.Run(fmt.Sprintf("%s/%d", backend.name, size), func(b *testing.B) {
				a, err := backend.new(16 * arena.RegionSize)
				if err != nil {
					b.Fatal(err)
				}
				defer a.Close()
				guard := a.Alloc(16)
				p := a.Alloc(size)
				a.Free(p)
				b.ReportAllocs()
				b.ResetTimer()
				for i := 0; i < b.N; i++ {
					p = a.Alloc(size)
					a.Free(p)
				}
				b.StopTimer()
				a.Free(guard)
			})
		}
	}
}

// A fresh arena per bounded burst lets the old implementation complete even for
// payloads it cannot recycle. This measures the whole cold allocation lifecycle,
// including map/close; ns/op refers to a burst of 256 objects, not one object.
func BenchmarkBurst(b *testing.B) {
	const count = 256
	for _, size := range []uint64{8192, 16384, 65536} {
		for _, backend := range backends {
			b.Run(fmt.Sprintf("%s/%d", backend.name, size), func(b *testing.B) {
				var offsets [count]uint64
				b.ReportAllocs()
				b.SetBytes(int64(count) * int64(size))
				for i := 0; i < b.N; i++ {
					a, err := backend.new(32 * arena.RegionSize)
					if err != nil {
						b.Fatal(err)
					}
					for j := range offsets {
						offsets[j] = a.Alloc(size)
						*(*uint64)(a.GetPtr(offsets[j])) = uint64(j)
					}
					for _, off := range offsets {
						a.Free(off)
					}
					if err := a.Close(); err != nil {
						b.Fatal(err)
					}
				}
			})
		}
	}
}

func BenchmarkBitmapChurn(b *testing.B) {
	a, err := arena.New(16 * arena.RegionSize)
	if err != nil {
		b.Fatal(err)
	}
	defer a.Close()
	guard := a.Alloc(16)
	p := a.Alloc(8192)
	a.Free(p)
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		p = a.Alloc(8192)
		a.Free(p)
	}
	b.StopTimer()
	s := a.Stats()
	b.ReportMetric(float64(s.HighWaterMark), "arena-high-water-B")
	a.Free(guard)
}

func BenchmarkParallelReuse(b *testing.B) {
	for _, backend := range backends {
		b.Run(backend.name, func(b *testing.B) {
			a, err := backend.new(32 * arena.RegionSize)
			if err != nil {
				b.Fatal(err)
			}
			defer a.Close()
			guard := a.Alloc(16)
			b.ReportAllocs()
			b.ResetTimer()
			b.RunParallel(func(pb *testing.PB) {
				for pb.Next() {
					p := a.Alloc(64)
					a.Free(p)
				}
			})
			b.StopTimer()
			a.Free(guard)
		})
	}
}

var lookupSink uint64

  b.ResetTimer();var sum uint64

func BenchmarkMissAfterGrowth(b *testing.B) {
 for _,regions:=range []uint64{16,512,2048}{
  for _,backend:=range backends { b.Run(fmt.Sprintf("%s/%d",backend.name,regions),func(b *testing.B){
   a,err:=backend.new((regions+4)*shared.RegionSize);if err!=nil{b.Fatal(err)};defer a.Close()
   for i:=uint64(0);i<regions;i++{a.AllocBatch(shared.RegionSize-shared.HeaderSize,1)}
   b.ReportAllocs();b.ResetTimer()
   for i:=0;i<b.N;i++{h,_,n:=a.PopBatch(0,1);if h!=0||n!=0{b.Fatal("expected empty class")}}
  })}
 }
}
func TestHistoricalFiles(t *testing.T) {
 for _,stable:=range []bool{false,true}{
  f,err:=os.CreateTemp(t.TempDir(),"historical");if err!=nil{t.Fatal(err)};defer f.Close()
  const size=8*shared.RegionSize
  if err=f.Truncate(size);err!=nil{t.Fatal(err)}
  var a allocator
  if stable{a,err=arena.Open(size,int(f.Fd()))}else{a,err=utils.NewArena[uint64](size,int(f.Fd()))}
  if err!=nil{t.Fatal(err)}
  p:=a.Alloc(8192);dead:=a.Alloc(64);a.Free(dead)
  *(*uint64)(a.GetPtr(p))=0xaabbccddeeff0011
  if err=a.Close();err!=nil{t.Fatal(err)}
  if stable{
   a,err:=nonmoving.Open(size,int(f.Fd()));if err!=nil{t.Fatal(err)}
   if *(*uint64)(a.GetPtr(p))!=0xaabbccddeeff0011{t.Fatal("old adaptive payload lost")}
   a.Free(p);if err=a.Trim();err!=nil{t.Fatal(err)}
   if a.Stats().ActiveClaims!=0{t.Fatal("old adaptive claims stranded")}
   if err=a.Close();err!=nil{t.Fatal(err)}
  }else{
   if a,err:=shared.NewArena[uint64](size,int(f.Fd()));err==nil{a.Close();t.Fatal("old utils mapping accepted")}
  }
 }
